 281     def add_arguments(self, parser):
 282 
 283         parser.add_argument(
 284             "--vendor-count",
 285             type=int,
 286             default=10,
 287             help="Number of vendors to create. Default: 10.",
 288         )
 289 
 290         parser.add_argument(
 291             "--stores-per-vendor",
 292             type=int,
 293             default=5,
 294             help="Number of stores per vendor. Default: 5.",
 295         )
 296 
 297         parser.add_argument(
 298             "--overwrite",
 299             action="store_true",
 300             help=(
 301                 "Delete previously generated mock vendors "
 302                 "before creating new ones."
 303             ),
 304         )

 310     def handle(self, *args, **options):
 311 
 312         vendor_count = options["vendor_count"]
 313         stores_per_vendor = options["stores_per_vendor"]
 314         overwrite = options["overwrite"]
 315 
 316         if vendor_count < 1:
 317 
 318             self.stdout.write(
 319                 self.style.ERROR(
 320                     "vendor-count must be at least 1."
 321                 )
 322             )
 323 
 324             return
 325 
 326         if stores_per_vendor < 1:
 327 
 328             self.stdout.write(
 329                 self.style.ERROR(
 330                     "stores-per-vendor must be at least 1."
 331                 )
 332             )
 333 
 334             return
 335 
 336         # ------------------------------------------------------
 337         # We need enough different locations.
 338         # ------------------------------------------------------
 339 
 340         if stores_per_vendor > len(self.LOCATIONS):
 341 
 342             self.stdout.write(
 343                 self.style.ERROR(
 344                     f"You requested {stores_per_vendor} stores "
 345                     f"per vendor, but only "
 346                     f"{len(self.LOCATIONS)} different locations "
 347                     f"are configured."
 348                 )
 349             )
 350 
 351             return
 352 
 353         # ------------------------------------------------------
 354         # Overwrite
 355         # ------------------------------------------------------
 356 
 357         if overwrite:
 358 
 359             self.stdout.write(
 360                 self.style.WARNING(
 361                     "Deleting previously generated "
 362                     "mock vendors..."
 363                 )
 364             )

 366             self._delete_existing_vendors()
 367 
 368             self.stdout.write("")

 370         # ------------------------------------------------------
 371         # Header
 372         # ------------------------------------------------------
 373 
 374         self.stdout.write(
 375             self.style.SUCCESS(
 376                 f"Creating {vendor_count} mock vendor(s)..."
 377             )
 378         )

 381             f"Each vendor will have "
 382             f"{stores_per_vendor} stores."

 386             "Each vendor's stores will be in "
 387             "different locations."

 602     def _get_unique_locations(
 603         self,
 604         number_of_stores,
 605     ):
 606         """
 607         Select distinct locations for ONE vendor.
 608 
 609         Example for 5 stores:
 610 
 611             Lagos
 612             Abuja
 613             Port Harcourt
 614             Ibadan
 615             Enugu
 616 
 617         The locations are sampled without replacement,
 618         therefore the same vendor can never receive two
 619         stores in the same configured location.
 620         """
 621 
 622         return random.sample(
 623             self.LOCATIONS,
 624             number_of_stores,
 625         )

 798     def _generate_unique_company_name(self):
 799 
 800         max_attempts = 100
 801 
 802         for _ in range(max_attempts):
 803 
 804             company_name = (
 805                 f"{random.choice(self.COMPANY_PREFIXES)} "
 806                 f"{random.choice(self.COMPANY_TYPES)}"
 807             )
 808 
 809             if not VendorProfile.objects.filter(
 810                 company_name=company_name
 811             ).exists():
 812 
 813                 return company_name
 814 
 815         return (
 816             "Solid Distributor "
 817             f"{random.randint(10000, 99999)}"
 818         )

 824     def _generate_unique_vendor_email(
 825         self,
 826         vendor_number,
 827     ):
 828 
 829         base_email = (
 830             f"vendor{vendor_number}"
 831             "@soliddistributor.com"
 832         )
 833 
 834         if not User.objects.filter(
 835             email=base_email
 836         ).exists():
 837 
 838             return base_email
 839 
 840         counter = 2
 841 
 842         while True:
 843 
 844             email = (
 845                 f"vendor{vendor_number}_{counter}"
 846                 "@soliddistributor.com"
 847             )
 848 
 849             if not User.objects.filter(
 850                 email=email
 851             ).exists():
 852 
 853                 return email
 854 
 855             counter += 1

 857     # ==========================================================
 858     # PHONE
 859     # ==========================================================
 860 
 861     def _generate_unique_phone(self):
 862 
 863         max_attempts = 1000
 864 
 865         for _ in range(max_attempts):
 866 
 867             prefix = random.choice(
 868                 self.NIGERIAN_PHONE_PREFIXES
 869             )
 870 
 871             subscriber_number = "".join(
 872                 str(random.randint(0, 9))
 873                 for _ in range(7)
 874             )
 875 
 876             phone = (
 877                 f"{prefix}{subscriber_number}"
 878             )
 879 
 880             # User.phone_number is unique.
 881             if User.objects.filter(
 882                 phone_number=phone
 883             ).exists():
 884 
 885                 continue
 886 
 887             return phone
 888 
 889         raise RuntimeError(
 890             "Unable to generate a unique "
 891             "Nigerian phone number."
 892         )

 898     def _generate_unique_store_slug(
 899         self,
 900         vendor,
 901         store_name,
 902     ):
 903 
 904         base_slug = slugify(
 905             store_name
 906         )
 907 
 908         slug = base_slug
 909         counter = 2
 910 
 911         while VendorStore.objects.filter(
 912             vendor=vendor,
 913             slug=slug,
 914         ).exists():
 915 
 916             slug = (
 917                 f"{base_slug}-{counter}"
 918             )
 919 
 920             counter += 1
 921 
 922         return slug

 950     def _randomize_coordinates(
 951         self,
 952         latitude,
 953         longitude,
 954     ):
 955         """
 956         Add a small coordinate variation around the
 957         configured location.
 958 
 959         This means two different vendors can have
 960         stores in the same city while still having
 961         slightly different GPS pickup points.
 962 
 963         However, stores belonging to the SAME vendor
 964         are always assigned different base locations.
 965         """
 966 
 967         latitude_offset = Decimal(
 968             str(
 969                 round(
 970                     random.uniform(
 971                         -0.015,
 972                         0.015,
 973                     ),
 974                     7,
 975                 )
 976             )
 977         )
 978 
 979         longitude_offset = Decimal(
 980             str(
 981                 round(
 982                     random.uniform(
 983                         -0.015,
 984                         0.015,
 985                     ),
 986                     7,
 987                 )
 988             )
 989         )
 990 
 991         latitude = (
 992             latitude + latitude_offset
 993         )
 994 
 995         longitude = (
 996             longitude + longitude_offset
 997         )
 998 
 999         latitude = max(
1000             Decimal("-90"),
1001             min(
1002                 Decimal("90"),
1003                 latitude,
1004             ),
1005         )
1006 
1007         longitude = max(
1008             Decimal("-180"),
1009             min(
1010                 Decimal("180"),
1011                 longitude,
1012             ),
1013         )
1014 
1015         return (
1016             latitude,
1017             longitude,
1018         )
